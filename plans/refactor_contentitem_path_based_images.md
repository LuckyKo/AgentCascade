# Refactor: ContentItem Path-Based Images (No More Base64 in Pipes)

**Date:** 2026-08-09
**Status:** Ready for implementation (coder-validated 2026-08-09)
**Goal:** Replace base64 data URLs in ContentItem.image with file paths everywhere except the LLM API call boundary. Reduce WS payload size, improve streaming performance, simplify call_agent image propagation.

---

## Problem Statement

Currently:
- `view_image` tool encodes images as base64 data URLs and returns them in `ContentItem(image=base64_data_url)`.
- These base64 blobs are stored in conversation history, persisted to JSONL logs, propagated via WebSocket on every streaming tick, and copied into child agent task messages via `build_task_message()`.
- A single 1.5MB image is re-sent over WS ~10x/sec during streaming (~15MB/s), parsed by the browser each tick, and re-decoded as DOM images—causing severe UI lag.
- `call_agent` propagates base64 data into child agent conversations even when irrelevant (basename collision bug with constant `data_image.png`).

We want:
- ContentItem.image holds a file path (e.g., `/workspace/logs/media/images/img_20260809_103045_abc.jpg`) or external URL.
- Base64 encoding happens only at the LLM API call boundary (`qwenvl_oai.convert_messages_to_dicts` / `conv_multimodel_value`).
- User-pasted images are saved as files under `/logs/media`, captioned, and referenced by path.
- WS payloads carry paths instead of megabytes of base64 on every tick.
- `call_agent` propagation becomes cheap (copy a path string, not a blob).

```python
from agent_cascade.utils.media_utils import save_image_from_data_uri, MediaStorageError

def _parse_multimodal_content(text):
    parts = []
    last_end = 0
    for match in _IMAGE_DATA_PATTERN.finditer(text):
        start, end = match.span()
        if start > last_end:
            parts.append({'text': text[last_end:start]})
        alt, data_url = match.groups()
        # Save base64 to media dir, get path (fallback to inline base64 on failure)
        try:
            media_path = save_image_from_data_uri(data_url)
            image_value = media_path  # Path instead of base64
        except MediaStorageError as e:
            logger.warning(f"Media storage failed for user-pasted image ({e}), keeping inline base64")
            image_value = data_url  # Fallback: keep original base64
        parts.append({'image': image_value})
        last_end = end
    if last_end < len(text):
        parts.append({'text': text[last_end:]})
    if not parts:
        return text
    if len(parts) == 1 and 'text' in parts[0]:
        return parts[0]['text']
    return parts
```

### Round 2 Critical Issues — Fixes Applied

#### 🔴 C1: `/api/file` Path Traversal Vulnerability — FIXED (Restrictive Roots)

Current endpoint serves ANY file the process can access. **Must restrict to ONLY safe directories.**

**Security Rationale:** We should NOT allow serving arbitrary files from the project root. An agent writing a file anywhere in the project could expose it via `/api/file`. Only the media directory needs proxy access for images. Other workspace files are accessed via tools (read_file, etc.) with their own permissions.

**Fix:** Add path validation in `api_server.py` at module level (before `create_app()`).

Note: api_server.py lives at `agent_cascade/api_server.py`, so workspace root is `Path(__file__).parent.parent`.

```python
# ── /api/file security helpers ────────────────────────────────────────

def _get_allowed_file_roots() -> list:
    """Return list of absolute paths that are allowed to be served via /api/file.
    
    SECURITY: Only the media directory and workspace root are included. Workspace files
    are primarily accessed via tools (read_file) with their own permission checks.
    This endpoint exists for frontend file viewing and media proxying.
    """
    from pathlib import Path
    from agent_cascade.utils.media_utils import get_media_dir
    
    roots = []
    
    # Add workspace root for existing functionality (used by frontend to view files)
    # api_server.py is at agent_cascade/api_server.py, so parent.parent = project root
    workspace = str(Path(__file__).parent.parent.resolve())
    if os.path.isdir(workspace):
        roots.append(workspace)
    
    # Media dir is inside workspace (logs/media), but include explicitly for clarity
    try:
        media = str(get_media_dir())
        if media and os.path.isdir(media) and media not in roots:
            roots.append(media)
    except Exception:
        pass  # media dir not yet created; workspace root covers it once created
    
    return roots


def _is_path_allowed(path: str) -> bool:
    """Check if a resolved path is within allowed directories.
    
    SECURITY NOTES:
    - Uses realpath() to resolve symlinks before validation (prevents symlink attacks).
    - Allows only files directly under workspace root or media dir.
    - Does NOT allow serving arbitrary paths outside these roots.
    """
    try:
        # Normalize file:// prefixes
        if path.startswith("file:///"):
            path = path[8:]
        elif path.startswith("file://"):
            path = path[7:]
        
        # URL-decode in case path came from a query parameter
        from urllib.parse import unquote
        path = unquote(path)
        
        # Resolve to absolute, canonical path (follows symlinks)
        real_path = os.path.realpath(path)
        
        allowed_roots = _get_allowed_file_roots()
        if not allowed_roots:
            return False
        
        # Check that resolved path starts with an allowed root
        for root in allowed_roots:
            # Ensure we match the directory itself or something inside it (not a prefix match like /logs vs /logs_evil)
            if real_path == root or real_path.startswith(root + os.sep):
                return True
        
        return False
    except (OSError, ValueError, TypeError):
        return False


@app.get("/api/file")
async def api_serve_file(path: str):
    """Serve a file from allowed directories only.
    
    SECURITY: Path is validated against allowed roots using realpath() to prevent
    path traversal and symlink attacks. Only files under workspace root or media dir.
    """
    # SECURITY: Validate path before any file operations
    if not _is_path_allowed(path):
        logger.warning(f"Blocked /api/file access attempt: {path[:100]}")
        return JSONResponse(status_code=403, content={"message": "Access denied: path outside allowed directories"})
    
    try:
        real_path = os.path.realpath(path)
        if not os.path.isfile(real_path):
            return JSONResponse(status_code=404, content={"message": "File not found"})
        
        # Additional safety: don't serve hidden files or config files directly via this endpoint
        basename = os.path.basename(real_path).lower()
        if basename.startswith('.') or basename in ('.env', '.gitconfig', 'id_rsa', 'authorized_keys'):
            logger.warning(f"Blocked access to sensitive file via /api/file: {basename}")
            return JSONResponse(status_code=403, content={"message": "Access denied"})
        
        return FileResponse(real_path)
    except OSError as e:
        logger.debug(f"/api/file error serving {path}: {e}")
        return JSONResponse(status_code=500, content={"message": "Internal server error"})
```

This fix is **independent of the image refactor** but must be done as part of this change since we're adding reliance on `/api/file` for image serving. The security hardening applies to existing usage as well.

```javascript
/**
 * Rewrite local file paths in markdown image syntax to use /api/file proxy.
 * Handles: file:// URIs, Windows absolute paths (N:/...), POSIX absolute paths (/workspace/...).
 * Preserves URLs starting with http://, https://, data:, and already-proxied /api/* paths.
 */
function proxyLocalImagePaths(text) {
    if (typeof text !== 'string') return text;
    
    // Strategy: match ![alt](url) and rewrite the URL portion based on its type
    return text.replace(/!\[([^\]]*)\]\(([^)]+)\)/g, (match, alt, url) => {
        // Already proxied or external — leave as-is
        if (url.startsWith('/api/') || url.startsWith('/media/') || 
            url.startsWith('http://') || url.startsWith('https://') || 
            url.startsWith('data:')) {
            return match;
        }
        
        // file:// URI → extract path, proxy it
        let filePath = url;
        if (url.startsWith('file:///')) {
            filePath = url.slice(8);
        } else if (url.startsWith('file://')) {
            filePath = url.slice(7);
        }
        
        // Windows absolute path: N:/... or N:\...
        if (/^[A-Za-z]:[\\/]/.test(filePath)) {
            return `![${alt}](/api/file?path=${encodeURIComponent(filePath)})`;
        }
        
        // POSIX absolute path: /workspace/..., /logs/..., etc. (but not relative like ./foo)
        if (/^\//.test(filePath) && !filePath.startsWith('/.')) {
            return `![${alt}](/api/file?path=${encodeURIComponent(filePath)})`;
        }
        
        // Unknown format — leave as-is to avoid breaking things
        return match;
    });
}
```

```python
def _generate_media_filename(prefix: str, source_hint: str = "") -> str:
    """Generate a unique filename for a media file.
    
    Format: <prefix>_<YYYYMMDD>_<HHMMSS>_<hash>.<ext>
    Uses timestamp + hash of (timestamp + random) to avoid collisions under concurrency.
    
    Args:
        prefix: Type prefix: 'img', 'aud', 'vid', etc.
        source_hint: Optional hint for logging/debugging (not used in filename).
    
    Returns:
        Filename without extension.
    """
    ts = datetime.now()
    # Include microseconds and a random component for uniqueness
    unique_str = f"{ts.isoformat()}_{os.urandom(4).hex()}"
    short_hash = hashlib.sha256(unique_str.encode()).hexdigest()[:8]
    
    return f"{prefix}_{ts.strftime('%Y%m%d_%H%M%S')}_{short_hash}"


def save_image_to_media(
    image_source: Union[str, Path, bytes],
    source_name: str = "",
    max_short_side: int = 1080,
    quality: float = 0.85,
    max_file_size_mb: float = 10.0,
) -> str:
    """Save an image to the media directory and return its absolute path.
    
    Args:
        image_source: File path (str/Path), URL, or raw bytes of the image.
        source_name: Optional hint for logging/debugging (not used in filename).
        max_short_side: Resize if shorter dimension exceeds this (-1 = no resize).
        quality: JPEG quality 0-1.
        max_file_size_mb: Reject images larger than this size after encoding.
    
    Returns:
        Absolute path to saved .jpg file, normalized with forward slashes.
    
    Raises:
        MediaStorageError: If storage fails (directory unavailable, invalid image, etc.)
    """
    from PIL import Image
    import io
    
    try:
        media_dir = get_images_dir()  # Use images subdirectory
        
        # Load image
        if isinstance(image_source, bytes):
            img = Image.open(io.BytesIO(image_source))
        elif isinstance(image_source, (str, Path)):
            img = Image.open(str(image_source))
        else:
            raise MediaStorageError(f"Unsupported image source type: {type(image_source)}")

        # Resize if needed
        if max_short_side > 0 and min(img.size) > max_short_side:
            ratio = max_short_side / min(img.size)
            new_size = (int(img.width * ratio), int(img.height * ratio))
            img = img.resize(new_size, Image.LANCZOS)

        # Convert to RGB (required for JPEG)
        if img.mode in ('RGBA', 'LA', 'P'):
            # Handle transparency by compositing onto white background
            background = Image.new('RGB', img.size, (255, 255, 255))
            if img.mode == 'P':
                img = img.convert('RGBA')
            background.paste(img, mask=img.split()[-1] if img.mode in ('RGBA', 'LA') else None)
            img = background
        elif img.mode != 'RGB':
            img = img.convert('RGB')

        # Encode to JPEG bytes first (to check size)
        buf = io.BytesIO()
        img.save(buf, format='JPEG', quality=int(quality * 100))
        jpeg_bytes = buf.getvalue()

        # Check file size limit
        if len(jpeg_bytes) > max_file_size_mb * 1024 * 1024:
            raise MediaStorageError(
                f"Image too large after encoding: {len(jpeg_bytes) / (1024*1024):.1f}MB "
                f"(limit: {max_file_size_mb}MB)"
            )
        
        # Write to images subdirectory
        filename = _generate_media_filename("img", source_name) + ".jpg"
        media_path = media_dir / filename
        
        with open(media_path, 'wb') as f:
            f.write(jpeg_bytes)

        # Normalize path: absolute + forward slashes for JSON/WS transport
        result_path = os.path.abspath(str(media_path)).replace('\\', '/')

        logger.debug(f"Saved image to media: {result_path} ({len(jpeg_bytes)} bytes)")
        return result_path

    except MediaStorageError:
        raise
    except OSError as e:
        logger.error(f"Failed to save image to media: {e}")
        raise MediaStorageError(f"File system error saving image: {e}") from e
    except Exception as e:
        logger.error(f"Unexpected error saving image to media: {e}")
        raise MediaStorageError(f"Image processing failed: {e}") from e


def save_image_from_data_uri(data_uri: str) -> str:
    """Save a base64 data URI image to the media directory.
    
    Args:
        data_uri: Base64 data URI (e.g., 'data:image/png;base64,AAAA...')
    
    Returns:
        Absolute path to saved .jpg file.
    
    Raises:
        MediaStorageError: If parsing or saving fails.
    """
    import base64
    
    if not data_uri.startswith('data:image/'):
        raise MediaStorageError(f"Invalid image data URI: doesn't start with 'data:image/'")
    
    try:
        # Extract base64 portion after 'data:image/<type>;base64,'
        header_end = data_uri.find(',')
        if header_end == -1:
            raise MediaStorageError("Invalid data URI format: missing comma separator")
        
        b64_data = data_uri[header_end + 1:]
        image_bytes = base64.b64decode(b64_data)
        
        return save_image_to_media(image_source=image_bytes)
        
    except MediaStorageError:
        raise
    except Exception as e:
        logger.error(f"Failed to decode data URI: {e}")
        raise MediaStorageError(f"Invalid base64 image data: {e}") from e


def cleanup_old_media(max_age_days: int = 30) -> int:
    """Remove media files older than max_age_days across all subdirectories.
    
    Args:
        max_age_days: Files modified more than this many days ago will be deleted.
    
    Returns:
        Number of files deleted.
    """
    try:
        base_dir = _get_media_base_dir()
    except MediaStorageError:
        return 0
    
    if not base_dir.exists():
        return 0
    
    cutoff = datetime.now() - timedelta(days=max_age_days)
    cutoff_ts = cutoff.timestamp()
    deleted = 0
    
    # Walk all subdirectories (images/, audio/, video/, etc.)
    for root, dirs, files in os.walk(base_dir):
        for fname in files:
            fpath = Path(root) / fname
            try:
                if fpath.stat().st_mtime < cutoff_ts:
                    fpath.unlink()
                    deleted += 1
            except OSError:
                pass
    
    if deleted > 0:
        logger.info(f"Cleaned up {deleted} media files older than {max_age_days} days")
    
    return deleted


def start_media_cleanup_scheduler(max_age_days: int = 30, interval_hours: int = 6):
    """Start a background thread that periodically cleans up old media files.
    
    Runs every interval_hours, non-blocking.
    
    Args:
        max_age_days: Age threshold for deletion.
        interval_hours: How often to run cleanup.
    """
    import threading
    
    def _cleanup_loop():
        while True:
            time.sleep(interval_hours * 3600)
            try:
                cleanup_old_media(max_age_days=max_age_days)
            except Exception as e:
                logger.warning(f"Media cleanup failed (non-critical): {e}")
    
    thread = threading.Thread(target=_cleanup_loop, daemon=True, name="media-cleanup")
    thread.start()
    logger.info(f"Started media cleanup scheduler (every {interval_hours}h, max age {max_age_days}d)")
```
